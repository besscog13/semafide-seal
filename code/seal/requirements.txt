# Dependencies for code/seal/ only.
#
# The package is deliberately standalone. It needs no numerical or web stack,
# so running the live code does not require compiling scipy or scikit-learn.
# CI asserts that isolation on every push rather than trusting this comment.
#
#   pip install -r code/seal/requirements.txt
#   cd code && python -m seal.demo
#
# The package is also installable directly, which is the preferred route:
#
#   pip install -e .

# The floor is what has actually been run, not what was inherited. The package
# uses InvalidSignature, hashes, serialization, and ec, all long-stable API, and
# the full suite plus the demo pass on 41.0.7. A higher floor would be a claim
# nobody checked, and it makes pip fail on any system where cryptography is
# managed by the distribution rather than by pip.
cryptography>=41.0.0

# Only for running the test suite.
pytest>=8.0.0

# Property-based testing. Test-only, like pytest. The live package imports
# neither, and CI asserts that on every push.
hypothesis>=6.0
